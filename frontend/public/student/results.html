function formatDate(iso) { const date = new Date(iso); return date.toLocaleDateString(undefined, { year: 'numeric', month: 'short', day: 'numeric' }); } function loadResults() { return JSON.parse(localStorage.getItem('smartexam_results') || '[]'); } function render() { const results = loadResults(); resultsBody.innerHTML = ''; if (!results.length) { emptyState.style.display = 'block'; return; } emptyState.style.display = 'none'; results.sort((a, b) => new Date(b.date) - new Date(a.date)).forEach(result => { const percent = result.totalMarks ? Math.round((result.score / result.totalMarks) * 100) : 0; const status = percent >= 50 ? 'completed' : 'missed'; const row = document.createElement('tr'); row.innerHTML = ` ${formatDate(result.date)} ${result.student || 'Student'} ${result.testName || '—'} ${result.score} / ${result.totalMarks} (${percent}%) ${result.durationMinutes ?? '—'}m ${status === 'completed' ? 'Completed ✅' : 'Needs improvement ⚠️'} `; resultsBody.appendChild(row); }); } function exportCsv() { const results = loadResults(); if (!results.length) { alert('No results to export yet.'); return; } const headers = ['Date', 'Student', 'Test', 'Score', 'Total Marks', 'Percentage', 'Duration', 'Status']; const rows = results.map(r => { const percent = r.totalMarks ? Math.round((r.score / r.totalMarks) * 100) : 0; const status = percent >= 50 ? 'Completed' : 'Needs improvement'; return [formatDate(r.date), r.student, r.testName, r.score, r.totalMarks, `${percent}%`, `${r.durationMinutes}m`, status]; }); const csvContent = [headers, ...rows].map(r => r.map(c => `"${String(c).replace(/"/g, '""')}"`).join(',')).join('\n'); const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' }); const url = URL.createObjectURL(blob); const link = document.createElement('a'); link.href = url; link.download = 'student-results.csv'; link.click(); URL.revokeObjectURL(url); } refreshBtn.addEventListener('click', e => { e.preventDefault(); render(); }); exportBtn.addEventListener('click', e => { e.preventDefault(); exportCsv(); }); // Auto-refresh every 2 seconds to show new results in real-time render(); setInterval(render, 2000);
